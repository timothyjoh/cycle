```bash
#!/usr/bin/env bash
# Selectively stage the cycle's intended change surface and create one
# commit. Honors a hard denylist (.claude, dist, node_modules, *.lock,
# 160000 gitlinks). Residual untracked/modified paths are reported on
# stderr but never staged. CYCLE_ID + CYCLE_TITLE come from the engine.
set -euo pipefail

: "${CYCLE_ID:?CYCLE_ID must be set by cycle engine}"
: "${CYCLE_TITLE:?CYCLE_TITLE must be set}"

is_denied() {
  local p="$1"
  case "$p" in
    .claude/*|.claude) return 0 ;;
    dist/*|dist) return 0 ;;
    node_modules/*|node_modules) return 0 ;;
    .cycle/cycle.pid) return 0 ;;
    *.lock) return 0 ;;
  esac
  return 1
}

# Pre-compute paths that are tracked as gitlinks (mode 160000).
gitlink_paths=$(git ls-files --stage 2>/dev/null | awk '$1 == "160000" { print $4 }' || true)
is_gitlink() {
  local p="$1"
  [ -n "$gitlink_paths" ] && printf '%s\n' "$gitlink_paths" | grep -Fxq -- "$p" && return 0
  [ -e "$p/.git" ] && return 0
  return 1
}

# Iterate git status --porcelain. Each line: XY<space><path>; for renames the
# path is "orig -> new" and we want only the new path.
while IFS= read -r line; do
  [ -z "$line" ] && continue
  xy="${line:0:2}"
  path="${line:3}"
  # Handle rename: "R  orig -> new" — keep new side.
  case "$xy" in
    R*|C*)
      path="${path##* -> }"
      ;;
  esac
  # Strip surrounding quotes git uses for paths with special chars.
  path="${path#\"}"
  path="${path%\"}"
  if is_denied "$path" || is_gitlink "$path"; then
    printf 'commit.sh: unstaged residual: %s\n' "$path" >&2
    continue
  fi
  git add -- "$path"
done < <(git status --porcelain)

if git diff --cached --quiet; then
  echo "commit.sh: nothing to commit"
  exit 0
fi
git commit -m "cycle ${CYCLE_ID}: ${CYCLE_TITLE}"
git rev-parse HEAD
```

**Notes on portability** (macOS bash 3.2 + Linux bash ≥ 4):
- Process substitution `< <(...)` is supported in 3.2.
- `${var:0:2}` substring and `${var:3}` offset slice are 3.2-safe.
- `grep -Fxq` is POSIX (BSD + GNU grep).
- No associative arrays.
